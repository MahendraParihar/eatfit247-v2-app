Authoring a new template

// templates/member/your-template.ejs
<%- await include('../layout/email-head', {
  subject: 'Your subject line',
  preheader: 'Inbox preview line — sets the tone.',
  franchise: franchise
}) %>

<%- await include('../partials/heading', {
  eyebrow: 'CATEGORY',
  title: 'Your H1 headline',
  franchise: franchise
}) %>

<p style="margin:0 0 16px 0;">Dear <%= memberName %>,</p>

<%- await include('../partials/section-open', { heading: 'Details', franchise: franchise }) %>
  <%- await include('../partials/info-row', { label: 'Field', value: yourValue, last: true }) %>
<%- await include('../partials/section-close') %>

<%- await include('../partials/button', { href: ctaUrl, label: 'Take action', franchise: franchise }) %>

<%- await include('../layout/email-foot', { franchise: franchise }) %>
